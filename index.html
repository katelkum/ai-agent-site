$ErrorActionPreference = 'SilentlyContinue'
[Console]::OutputEncoding = [System.Text.Encoding]::UTF8

$desktop = [Environment]::GetFolderPath('Desktop')
$reportPath = "$desktop\AUDIT_COMPLETO_SISTEMA_100.txt"

Clear-Host
Write-Host "========================================================================" -ForegroundColor Cyan
Write-Host "   AUDIT TOTALE FORENSE DEL COMPUTER (ENTERPRISE EDITION - 100% SICURO) " -ForegroundColor Cyan
Write-Host "========================================================================" -ForegroundColor Cyan
Write-Host "Raccolta di tutti i parametri hardware e software in corso (circa 25s)...`n" -ForegroundColor Yellow

$out = [System.Collections.Generic.List[string]]::new()
function Log($txt) { $script:out.Add($txt) }

Log("================================================================================")
Log("          REPORT FORENSE COMPLETO A 360 GRADI - STATO TOTALE DEL PC             ")
Log("              Data di generazione: $(Get-Date -Format 'dd/MM/yyyy HH:mm:ss')    ")
Log("================================================================================`n")

# [1] SISTEMA OPERATIVO, CPU E STATO RIAVVIO
Write-Host "[1/13] Analisi architettura, CPU, Uptime e riavvii in sospeso..." -ForegroundColor Gray
$os = Get-CimInstance Win32_OperatingSystem
$cs = Get-CimInstance Win32_ComputerSystem
$up = (Get-Date) - $os.LastBootUpTime
$rebootPending = (Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending") -or (Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired")
$pendingFiles = (Get-ItemProperty "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager" -Name "PendingFileRenameOperations" -ErrorAction SilentlyContinue).PendingFileRenameOperations
Log("--- [1] ARCHITETTURA, SISTEMA E STATO BLOCCO RIAVVIO ---")
Log("Computer: $($cs.Name) | Produttore: $($cs.Manufacturer) | Modello: $($cs.Model)")
Log("Windows: $($os.Caption) - Versione/Build: $($os.BuildNumber).$($os.Version) (Architettura: $($os.OSArchitecture))")
Log("Processore: $((Get-CimInstance Win32_Processor).Name) [Core: $((Get-CimInstance Win32_Processor).NumberOfCores) | Thread: $((Get-CimInstance Win32_Processor).NumberOfLogicalProcessors)]")
Log("Tempo di Accensione Continuo: $($up.Days) giorni, $($up.Hours) ore, $($up.Minutes) minuti")
Log("Riavvio di sistema obbligatorio in sospeso: $rebootPending")
Log("File/Driver di sistema bloccati in attesa di riavvio (Pending Renames): $(if($pendingFiles){'PRESENTI (Modifiche in coda)'}else{'NESSUNO [Pulito]'})`n")

# [2] MEMORIA RAM, PAGING, LEAK HANDLES E KERNEL POOL
Write-Host "[2/13] Scansione chirurgica RAM, File di Paging e perdite Kernel..." -ForegroundColor Gray
$totR = [math]::Round($os.TotalVisibleMemorySize / 1MB, 2)
$freeR = [math]::Round($os.FreePhysicalMemory / 1MB, 2)
$usedR = [math]::Round($totR - $freeR, 2)
Log("--- [2] MEMORIA RAM, MEMORIA VIRTUALE E CONSUMO POOL ---")
Log("RAM Totale: $totR GB | In Uso: $usedR GB | Libera: $freeR GB")
Log("Paging Automatico: $($cs.AutomaticManagedPagefile)")
$pfiles = Get-CimInstance Win32_PageFileUsage
if ($pfiles) { foreach ($pf in $pfiles) { Log("File Paging Attivo: $($pf.Name) ($($pf.AllocatedBaseSize) MB allocati, $($pf.CurrentUsage) MB usati)") } }
else { Log("AVVISO: File di paging disabilitato o non rilevato!") }

Log("`nTop 5 Processi per consumo RAM e percorso di avvio:")
Get-Process | Sort-Object WorkingSet64 -Descending | Select-Object -First 5 ProcessName, Id, @{N='RAM_MB';E={[math]::Round($_.WorkingSet64/1MB,1)}}, Handles, @{N='Percorso';E={$_.Path}} | Out-String | ForEach-Object { Log($_) }

Log("Top 5 Processi con piu' Handle (Rilevamento Memory Leak):")
Get-Process | Sort-Object Handles -Descending | Select-Object -First 5 ProcessName, Id, Handles, @{N='RAM_MB';E={[math]::Round($_.WorkingSet64/1MB,1)}} | Out-String | ForEach-Object { Log($_) }

Log("Top 5 Processi per consumo Pool Non di Paging (NPM Leak):")
Get-Process | Sort-Object NPM -Descending | Select-Object -First 5 ProcessName, Id, @{N='NPM_MB';E={[math]::Round($_.NPM/1MB,2)}} | Out-String | ForEach-Object { Log($_) }

# [3] SALUTE HARDWARE SSD, USURA, TEMPERATURA E INTEGRITA'
Write-Host "[3/13] Interrogazione contatori fisici SMART, usura SSD e TRIM..." -ForegroundColor Gray
Log("--- [3] HARDWARE DISCHI SSD / NVMe E PARAMETRI SMART ---")
$pDisks = Get-PhysicalDisk
foreach ($pd in $pDisks) {
    Log("Disco #$($pd.DeviceId): $($pd.FriendlyName) [Tipo: $($pd.MediaType) - Bus: $($pd.BusType) - Salute: $($pd.HealthStatus) - Stato: $($pd.OperationalStatus)]")
    $rel = Get-StorageReliabilityCounter -PhysicalDisk $pd
    if ($rel) { Log("  -> Temperatura: $($rel.Temperature)C | Percentuale Usura: $($rel.Wear)% | Errori Lettura: $($rel.ReadErrorsTotal) | Errori Scrittura: $($rel.WriteErrorsTotal) | Ore Accensione: $([math]::Round($rel.PowerOnHours.TotalHours,1))") }
}
Log("Stato TRIM: $(fsutil behavior query DisableDeleteNotify)")
Log("Integrita' Filesystem Partizione C: : $(fsutil dirty query C:)`n")

# [4] MISURAZIONE ATTIVITA' DISCO IN TEMPO REALE
Write-Host "[4/13] Rilevamento in tempo reale dei processi che scrivono su disco..." -ForegroundColor Gray
Log("--- [4] PROCESSI CON MAGGIORE ATTIVITA' DISCO E CPU ADESSO ---")
$topIO = Get-CimInstance Win32_PerfFormattedData_PerfProc_Process | Where-Object { $_.Name -notmatch "_Total|Idle" } | Sort-Object IODataBytesPersec -Descending | Select-Object -First 5 Name, @{N='Disco_MB_s';E={[math]::Round($_.IODataBytesPersec/1MB,2)}}, PercentProcessorTime
$topIO | Out-String | ForEach-Object { Log($_) }

# [5] SCHEDA VIDEO, GRAFICA E CRASH DISPLAY
Write-Host "[5/13] Verifica driver GPU, VRAM e stabilita' video..." -ForegroundColor Gray
Log("--- [5] GRAFICA, DISPLAY E DRIVER VIDEO ---")
$gpus = Get-CimInstance Win32_VideoController
foreach ($gpu in $gpus) {
    Log("GPU: $($gpu.Name) - Risoluzione: $($gpu.CurrentHorizontalResolution)x$($gpu.CurrentVerticalResolution) @ $($gpu.CurrentRefreshRate)Hz")
    Log("  -> Versione Driver: $($gpu.DriverVersion) Data: $($gpu.DriverDate.ToString('dd/MM/yyyy')) | VRAM: $([math]::Round($gpu.AdapterRAM/1MB)) MB")
}
$gpuCrashes = Get-WinEvent -FilterHashtable @{LogName='System'; ProviderName='Display'; StartTime=(Get-Date).AddDays(-7)} -MaxEvents 3
if ($gpuCrashes) {
    Log("AVVISO: Rilevati crash recenti del driver video (TDR):")
    foreach ($c in $gpuCrashes) { Log("  Data: $($c.TimeCreated) - $($c.Message)") }
} else { Log("Nessun crash del driver video rilevato negli ultimi 7 giorni. [OK]`n") }

# [6] WI-FI, SCHEDE DI RETE E LATENZA INTERNET
Write-Host "[6/13] Test connettivita' Wi-Fi, DNS e latenza di rete..." -ForegroundColor Gray
Log("--- [6] RETE, WI-FI E LATENZA INTERNET ---")
Get-NetAdapter | Select-Object Name, InterfaceDescription, Status, LinkSpeed | Out-String | ForEach-Object { Log($_) }
$wlan = netsh wlan show interfaces
$ssid = ($wlan | Select-String "^\s*SSID\s*:\s*(.+)$").Matches.Groups[1].Value.Trim()
$signal = ($wlan | Select-String "^\s*Segnale\s*:\s*(.+)$|^\s*Signal\s*:\s*(.+)$").Matches.Groups[1].Value.Trim()
$rate = ($wlan | Select-String "^\s*Velocit.*ricezione.*:\s*(.+)$|^\s*Receive rate.*:\s*(.+)$").Matches.Groups[1].Value.Trim()
Log("Wi-Fi Attuale: SSID: $ssid | Segnale: $signal | Link: $rate")

$pingGateway = Test-Connection -ComputerName 1.1.1.1 -Count 2
$pingDNS = Test-Connection -ComputerName google.com -Count 2
if ($pingGateway) {
    $avgPing = [math]::Round(($pingGateway | Measure-Object -Property ResponseTime -Average).Average, 1)
    Log("Connessione IP Esterno (1.1.1.1): FUNZIONANTE [OK] (Latenza media: $avgPing ms)")
} else { Log("Connessione IP Esterno: FALLITA [Nessun accesso a Internet]") }

if ($pingDNS) { Log("Risoluzione Nomi DNS (google.com): FUNZIONANTE [OK]") }
else { Log("Risoluzione Nomi DNS: ERRORE DNS [Impossibile tradurre i domini web]") }
Log("")

# [7] BLUETOOTH E STATO SERVIZI
Write-Host "[7/13] Verifica modulo Bluetooth e servizi associati..." -ForegroundColor Gray
Log("--- [7] STATO BLUETOOTH ---")
$btDevices = Get-PnpDevice -Class Bluetooth
if ($btDevices) { $btDevices | Select-Object FriendlyName, Status, Present | Out-String | ForEach-Object { Log($_) } }
else { Log("Nessun dispositivo o modulo Bluetooth rilevato.") }
$btService = Get-Service -Name bthserv
Log("Servizio di supporto Bluetooth (bthserv): $($btService.Status) (Tipo Avvio: $($btService.StartType))`n")

# [8] SCHEDA AUDIO E MICROFONI
Write-Host "[8/13] Controllo periferiche audio e servizi multimediali..." -ForegroundColor Gray
Log("--- [8] AUDIO E MICROFONI ---")
Get-CimInstance Win32_SoundDevice | Select-Object Name, Status, Manufacturer | Out-String | ForEach-Object { Log($_) }
$audioSvc = Get-Service -Name Audiosrv
Log("Servizio Audio di Windows: $($audioSvc.Status) (Tipo Avvio: $($audioSvc.StartType))`n")

# [9] BATTERIA UNIVERSALE (PARSER XML UFFICIALE - FUNZIONA SU QUALSIASI MARCA)
Write-Host "[9/13] Analisi profonda batteria (Capacita' di fabbrica, usura % e cicli)..." -ForegroundColor Gray
Log("--- [9] SALUTE FISICA DELLA BATTERIA (UNIVERSALE) ---")
$bXmlPath = "$env:TEMP\bat_audit_temp.xml"
powercfg /batteryreport /xml /output $bXmlPath | Out-Null
if (Test-Path $bXmlPath) {
    try {
        [xml]$bXml = Get-Content $bXmlPath
        $design = [double]$bXml.BatteryReport.Batteries.Battery.DesignCapacity
        $full = [double]$bXml.BatteryReport.Batteries.Battery.FullChargeCapacity
        $cycles = $bXml.BatteryReport.Batteries.Battery.CycleCount
        if ($design -gt 0) {
            $wear = [math]::Round((1 - ($full / $design)) * 100, 1)
            Log("Capacita' di Fabbrica: $design mWh | Carica Massima Attuale: $full mWh")
            Log("Usura Reale Batteria: $wear % | Cicli di ricarica completati: $cycles")
        } else { Log("Dispositivo fisso (Desktop) o nessuna batteria interna installata.") }
    } catch { Log("Impossibile analizzare i parametri XML della batteria.") }
    Remove-Item $bXmlPath -Force -ErrorAction SilentlyContinue
} else { Log("Dispositivo fisso (Desktop) o batteria non supportata.") }
Log("")

# [10] CONFLITTI HARDWARE IN GESTIONE DISPOSITIVI
Write-Host "[10/13] Scansione conflitti hardware e punti esclamativi gialli..." -ForegroundColor Gray
Log("--- [10] CONFLITTI HARDWARE E DRIVER IN ERRORE (DEVICE MANAGER) ---")
$errPnp = Get-CimInstance Win32_PnPEntity | Where-Object { $_.ConfigManagerErrorCode -ne 0 -and $_.ConfigManagerErrorCode -ne $null }
if ($errPnp) {
    Log("ATTENZIONE: Trovate periferiche con errore hardware/driver:")
    $errPnp | Select-Object Name, DeviceID, ConfigManagerErrorCode | Out-String | ForEach-Object { Log($_) }
} else { Log("Nessun conflitto hardware trovato. Tutte le periferiche funzionano correttamente. [OK]`n") }

# [11] ATTIVITA' PIANIFICATE PARASSITE (SCHEDULED TASKS NON-MICROSOFT)
Write-Host "[11/13] Ricerca attivita' pianificate di terze parti e adware nascosti..." -ForegroundColor Gray
Log("--- [11] ATTIVITA' PIANIFICATE DI TERZE PARTI ATTIVE (SCHEDULED TASKS) ---")
$tasks = Get-ScheduledTask | Where-Object { $_.TaskPath -notmatch '\\Microsoft\\Windows\\' -and $_.State -ne 'Disabled' } | Select-Object TaskName, TaskPath, State
$tasks | Out-String | ForEach-Object { Log($_) }

# [12] ERRORI CRITICI, CRASH DUMP E FALLIMENTI WINDOWS UPDATE (ULTIMI 7 GIORNI)
Write-Host "[12/13] Analisi scatola nera (Crash BSOD e fallimenti Windows Update)..." -ForegroundColor Gray
Log("--- [12] ERRORI CRITICI, SCHERMATE BLU E WINDOWS UPDATE ---")
if (Test-Path "C:\Windows\Minidump\*.dmp") {
    $dumps = Get-ChildItem "C:\Windows\Minidump\*.dmp" | Sort-Object LastWriteTime -Descending | Select-Object -First 3
    Log("ATTENZIONE: Trovati file di crash (BSOD) recenti:")
    foreach ($d in $dumps) { Log("  File: $($d.Name) del $($d.LastWriteTime)") }
} else { Log("Nessun crash di sistema BSOD registrato. [OK]") }

$wuErrors = Get-WinEvent -FilterHashtable @{LogName='System'; ProviderName='Microsoft-Windows-WindowsUpdateClient'; Id=20; StartTime=(Get-Date).AddDays(-7)} -MaxEvents 3
if ($wuErrors) {
    Log("AVVISO: Rilevati aggiornamenti di Windows falliti negli ultimi 7 giorni:")
    foreach ($w in $wuErrors) { Log("  $($w.TimeCreated) -> $($w.Message)") }
} else { Log("Nessun fallimento critico recente registrato da Windows Update. [OK]") }
Log("")

# [13] SICUREZZA, DISM E PROGRAMMI ALL'AVVIO
Write-Host "[13/13] Verifica antivirus, integrita' archivio componenti e avvio..." -ForegroundColor Gray
Log("--- [13] SICUREZZA, STATO DISM E PROGRAMMI IN AVVIO ---")
$av = Get-CimInstance -Namespace root\SecurityCenter2 -ClassName AntiVirusProduct
if ($av) { foreach ($a in $av) { Log("Antivirus Rilevato: $($a.displayName)") } }
Log("Integrita' Component Store Windows: $(dism /Online /Cleanup-Image /CheckHealth)")

Log("`nServizi Critici di Sistema (Update / DoSvc / SysMain):")
Get-Service -Name wuauserv, DoSvc, SysMain, bits, trustedinstaller | Select-Object Name, Status, StartType | Out-String | ForEach-Object { Log($_) }

Log("Programmi registrati in Avvio Automatico (Run / Startup):")
Get-CimInstance Win32_StartupCommand | Select-Object Name, Command | Out-String | ForEach-Object { Log($_) }

# ==============================================================================
# PROMPT DEFINITIVO BLINDATO PER L'INTELLIGENZA ARTIFICIALE

Log("`n================================================================================")
Log("                ISTRUZIONI PER L'ASSISTENTE INTELLIGENZA ARTIFICIALE            ")
Log("================================================================================")

Write-Host "جاري جمع بيانات العتاد والبرامج بدقة (انتظر حوالي 25 ثانية)...`n" -ForegroundColor Yellow

Write-Host "[2/13] فحص استهلاك الرامات والذاكرة الافتراضية والتسريبات..." -ForegroundColor Gray
